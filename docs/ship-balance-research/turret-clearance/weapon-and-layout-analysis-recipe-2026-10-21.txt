import json,sys,math
from pathlib import Path
import numpy as np
ROOT=Path.cwd();sys.path.insert(0,str(ROOT/'scripts/ship-balance'))
import measure_turret_clearance as m
from UnityPy.helpers.MeshHelper import MeshHandler
OUT=ROOT/'docs/ship-balance-research/turret-clearance';TMP=ROOT/'.tmp/mothership-fix'
old=m.read(OUT/'measurements.json')
env=m.UnityPy.load(str(m.DEFAULT_GAME/'TerraInvicta_Data/StreamingAssets/AssetBundles/shipweapons'))
paths={Path(path).stem.lower():(path,ptr) for path,ptr in env.container.items()}
result=[];arrays={}
for k,g in enumerate(old['geometries']):
 resource=g['resource'];path,ptr=paths[resource.split('/')[-1].lower()]
 go=ptr.read();root=m.transform(go).read();child=root.m_Children[0].read()
 base,bname=m.mesh_vertices(go);upper,uname=m.mesh_vertices(child.m_GameObject.read())
 scale=m.xyz(root.m_LocalScale);cs=m.xyz(child.m_LocalScale);p=m.xyz(child.m_LocalPosition)
 upper_scaled=upper*cs
 r=m.local_matrix(child)[:3,:3]@np.diag(1/cs)
 base_points=base*scale
 upper_points=m.transform_points(m.local_matrix(child),upper)*scale
 meshes=[]
 for index,(partgo,points) in enumerate([(go,base_points),(child.m_GameObject.read(),upper_points)]):
  mesh=next(q for q in m.component_ptrs(partgo) if q.type.name=='MeshFilter').read().m_Mesh.read()
  handler=MeshHandler(mesh);handler.process();tris=np.concatenate([np.asarray(t,dtype=np.int32) for t in handler.get_triangles() if t])
  arrays['v%d_%d'%(k,index)]=points;arrays['t%d_%d'%(k,index)]=tris
  meshes.append({'name':mesh.m_Name,'key':'%d_%d'%(k,index)})
 radius=float(np.linalg.norm(base_points[:,[0,2]],axis=1).max())
 sampled=radius
 for deg in range(-180,181):
  a=math.radians(deg);co,si=math.cos(a),math.sin(a)
  pitch=np.array([[1,0,0],[0,co,-si],[0,si,co]])
  points=(p+upper_scaled@(r@pitch).T)*scale
  sampled=max(sampled,float(np.linalg.norm(points[:,[0,2]],axis=1).max()))
 # Every angle is within 0.5 degrees of a sample. Bound the missing arc displacement.
 epsilon=2*max(abs(scale))*float(np.linalg.norm(upper_scaled,axis=1).max())*math.sin(math.radians(.25))
 ws=[w for w in old['weapons'] if w['resource']==resource]
 row={'resource':resource,'asset_path':path,'meshes':meshes,'sizes':sorted({w['size'] for w in ws}),
      'weapons':[w['name'] for w in ws], 'root_scale':scale.tolist(),'child_scale':cs.tolist(),
      'child_pivot_m':(p*scale).tolist(),'source_child_rotation_xyzw':[child.m_LocalRotation.x,child.m_LocalRotation.y,child.m_LocalRotation.z,child.m_LocalRotation.w],
      'posed_dimensions_m':g['posed_dimensions_m'],'posed_min_m':g['posed_min_m'],'posed_max_m':g['posed_max_m'],
      'base_dimensions_m':np.ptp(base_points,axis=0).tolist(),'moving_dimensions_m':np.ptp(upper_points,axis=0).tolist(),
      'posed_yaw_diameter_m':2*g['yaw_only_radius_m'],'pitch_yaw_diameter_bound_m':2*(sampled+epsilon),
      'sampling_radius_margin_m':epsilon,'old_sphere_diameter_m':2*g['swept_radius_m'],
      'static_launcher':all(w['kind']=='TIMissileTemplate' for w in ws)}
 result.append(row)
 print(resource,'dimensions',*[round(x,3) for x in g['posed_dimensions_m']],'sphere',round(row['old_sphere_diameter_m'],3),'pitch/yaw',round(row['pitch_yaw_diameter_bound_m'],3),flush=True)
np.savez_compressed(TMP/'weapon-scenes.npz',**arrays)
(OUT/'weapon-mesh-inspection-2026-10-21.json').write_text(json.dumps(result,indent=2)+'\n',encoding='utf-8')

import json,math
from pathlib import Path

for i,w in enumerate(ws):

 x,y,z,qw=np.array(w['source_child_rotation_xyzw'])/np.linalg.norm(w['source_child_rotation_xyzw'])
 R=np.array([[1-2*(y*y+z*z),2*(x*y-z*qw),2*(x*z+y*qw)],
             [2*(x*y+z*qw),1-2*(x*x+z*z),2*(y*z-x*qw)],
             [2*(x*z-y*qw),2*(y*z+x*qw),1-2*(x*x+y*y)]])
 child=(upper/S-p)@R
 np.testing.assert_allclose(np.ptp(np.concatenate([base,upper]),axis=0),w['posed_dimensions_m'],rtol=0,atol=1e-6)
 for angle in rng.uniform(-math.pi,math.pi,20):
  c,s=math.cos(angle),math.sin(angle);P=np.array([[1,0,0],[0,c,-s],[0,s,c]])
  points=(p+child@(R@P).T)*S
  radius=max(np.linalg.norm(points[:,[0,2]],axis=1).max(),np.linalg.norm(base[:,[0,2]],axis=1).max())
  assert radius<=w['pitch_yaw_diameter_bound_m']/2+1e-7,(w['resource'],radius)
  checks+=1
