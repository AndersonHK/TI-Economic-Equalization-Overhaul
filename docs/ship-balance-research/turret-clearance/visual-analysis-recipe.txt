Visual-study analysis recipe (research archive; not a gameplay patch).
Dependencies: existing Python 3.12 + UnityPy/NumPy/Pillow; Blender 2.79 CPU background renderer.
Commands: export_geometry.py; render_hulls.py; render_hulls.py -- --details; surface_probe.py; export_materials.py; render_hulls.py -- --details --textured AlienMothership; annotate.py; proposal_evidence.py; write_report.py.
Run Python scripts from repository root. Blender invocation: blender.exe --background --factory-startup -t 8 --python SCRIPT -- OPTIONS. Temporary artifacts use .tmp/turret-visual; retained outputs use docs/ship-balance-research/turret-clearance.

===== export_geometry.py =====

import sys, json
from pathlib import Path
import numpy as np
ROOT=Path.cwd()
sys.path.insert(0,str(ROOT/'scripts/ship-balance'))
import measure_turret_clearance as m
from UnityPy.helpers.MeshHelper import MeshHandler
import generate_hull_variant_report as h

OUT=ROOT/'.tmp/turret-visual'
OUT.mkdir(parents=True,exist_ok=True)
game=m.DEFAULT_GAME
envs={'base':m.UnityPy.load(str(game/'TerraInvicta_Data/StreamingAssets/AssetBundles/ships')),
      'dlc':m.UnityPy.load(str(game/'DLC_Content/DarkSkies/AssetBundles/ships_prm'))}
evidence=m.read(ROOT/'docs/ship-balance-research/turret-clearance/measurements.json')
names={'Escort','Frigate','Monitor','Battleship','Dreadnought','Titan','AlienDreadnought','AlienAssaultCarrier','AlienMothership','AlienDestroyer'}
metadata=[]
for v in evidence['variants']:
    if v['hull'] not in names: continue
    env=envs['dlc' if v['appearance']>=2 and not v['alien'] else 'base']
    root=env.container[v['asset_path']].read()
    root_t=m.transform(root)
    origin=m.xyz(root_t.read().m_LocalPosition)
    records={'meshes':[],'colliders':[]}
    m.hull_reader.walk(root_t,np.eye(4),True,[],records)
    main,excluded=h.select_main_hull_meshes(m.hull_reader,v['hull'],records)
    meshes=[]; arrays={}
    for i,r in enumerate(main):
        handler=MeshHandler(r['mesh']); handler.process()
        vertices=np.asarray(handler.m_Vertices,dtype=float)
        tris=[np.asarray(t,dtype=np.int32) for t in handler.get_triangles() if t]
        if not tris:continue
        triangles=np.concatenate(tris)
        world=m.transform_points(r['matrix'],vertices)-origin
        arrays['v'+str(i)]=world; arrays['t'+str(i)]=triangles
        meshes.append({'key':i,'path':r['path'],'name':r['mesh_name'],'vertices':len(vertices),'triangles':len(triangles)})
    # Inventory all renderer components, including inactive children and turret placeholders.
    renderers=[]
    for goid,(matrix,tr,path) in m.hierarchy(root_t):
        go=tr.m_GameObject.read()
        for p in m.component_ptrs(go):
            if p.type.name in ('MeshRenderer','SkinnedMeshRenderer'):
                renderers.append({'path':path,'component':p.type.name,'active_self':bool(go.m_IsActive)})
    key=v['hull']+'-'+str(v['appearance'])
    np.savez_compressed(OUT/(key+'.npz'),**arrays)
    data=dict(v,meshes=meshes,renderer_inventory=renderers,geometry_file=key+'.npz')
    metadata.append(data)
    print(key,len(meshes),sum(r['triangles'] for r in meshes),flush=True)
(OUT/'scene-index.json').write_text(json.dumps(metadata,indent=2),encoding='utf-8')

# Blender 2.79 / Python 3.5. Offline clay render of decoded game geometry.
import bpy, json, math, sys
from pathlib import Path
import numpy as np
from mathutils import Vector, Quaternion
from bpy_extras.object_utils import world_to_camera_view

ROOT=Path.cwd(); TMP=ROOT/'.tmp/turret-visual'; OUT=ROOT/'docs/ship-balance-research/turret-clearance/visuals'
OUT.mkdir(parents=True,exist_ok=True)
scenes=json.loads((TMP/'scene-index.json').read_text())
args=sys.argv[sys.argv.index('--')+1:] if '--' in sys.argv else []
details='--details' in args
textured='--textured' in args
args=[a for a in args if a not in ['--details','--textured']]
if args:scenes=[v for v in scenes if v['hull'] in args]
render_meta=[]

def cv(p):return Vector((p[0],p[2],p[1]))

for v in scenes:
    bpy.ops.object.select_all(action='SELECT'); bpy.ops.object.delete(use_global=False)
    for mesh in list(bpy.data.meshes):
        if mesh.users==0:bpy.data.meshes.remove(mesh)
    scene=bpy.context.scene
    scene.render.engine='BLENDER_RENDER'
    scene.render.resolution_x=1600;scene.render.resolution_y=700;scene.render.resolution_percentage=100
    scene.render.image_settings.file_format='PNG'; scene.render.alpha_mode='SKY'
    scene.world.horizon_color=(.025,.032,.045)
    scene.world.ambient_color=(0,0,0)
    scene.world.light_settings.use_ambient_occlusion=True
    scene.world.light_settings.ao_factor=.65
    scene.world.light_settings.samples=5
    scene.world.light_settings.use_environment_light=True
    scene.world.light_settings.environment_energy=.12
    mat=bpy.data.materials.get('HullClay') or bpy.data.materials.new('HullClay')
    mat.diffuse_color=(.42,.48,.54);mat.diffuse_intensity=.8;mat.specular_intensity=.1
    data=np.load(str(TMP/v['geometry_file']))
    texture_info=json.loads((TMP/'material-index.json').read_text()) if textured else {}
    uvdata=np.load(str(TMP/'mothership-uv.npz')) if textured else None
    allpoints=[]
    for mr in v['meshes']:
        verts=data['v'+str(mr['key'])]; faces=data['t'+str(mr['key'])]
        # Swap Unity Y/Z and reverse triangle winding: a handedness conversion.
        verts=verts[:,[0,2,1]];faces=faces[:,::-1]
        mesh=bpy.data.meshes.new(mr['name']);mesh.from_pydata(verts.tolist(),[],faces.tolist());mesh.update()
        ob=bpy.data.objects.new(mr['name'],mesh);scene.objects.link(ob)
        if mr['name'] in texture_info:
            texmat=bpy.data.materials.new(mr['name']+' albedo');texmat.diffuse_color=(1,1,1);texmat.specular_intensity=.1
            tx=bpy.data.textures.new(mr['name']+' albedo','IMAGE')
            tx.image=bpy.data.images.load(str(TMP/texture_info[mr['name']]['_MainTex']['file']))
            slot=texmat.texture_slots.add();slot.texture=tx;slot.texture_coords='UV';slot.use_map_color_diffuse=True
            mesh.uv_textures.new(name='UVMap');layer=mesh.uv_layers.active.data
            uvs=uvdata[mr['name']]
            for polygon in mesh.polygons:
                for li in polygon.loop_indices:layer[li].uv=uvs[mesh.loops[li].vertex_index][:2]
            ob.data.materials.append(texmat)
        else:ob.data.materials.append(mat)
        allpoints.append(verts)
    points=np.concatenate(allpoints)
    lo=points.min(0);hi=points.max(0);center=(lo+hi)/2; extent=float(max(hi-lo))
    for i,(loc,energy) in enumerate([((1,-1,2),.7),((-1,.6,-1),.25)]):
        lamp=bpy.data.lamps.new('Key'+str(i),'SUN');lamp.energy=energy;lamp.use_shadow=False
        ob=bpy.data.objects.new('Key'+str(i),lamp);scene.objects.link(ob)
        ob.rotation_euler=Vector(loc).to_track_quat('Z','Y').to_euler()
    camera=bpy.data.cameras.new('Camera');cam=bpy.data.objects.new('Camera',camera);scene.objects.link(cam);scene.camera=cam
    camera.type='ORTHO';camera.clip_end=10000
    views=[('dorsal',(0,0,1),None,None),('side',(-1,0,0),None,None),('ventral',(0,0,-1),None,None)]
    if details:
        import itertools
        ns=[n for n in v['nodes'] if n['side']=='dorsal' and n['ordinary']]
        pairs=[(np.linalg.norm(np.asarray(a['position_m'])-b['position_m']),a,b) for a,b in itertools.combinations(ns,2)]
        if not pairs:continue
        dist,a,b=min(pairs,key=lambda x:x[0]);focus=(np.asarray(a['position_m'])+b['position_m'])/2
        views=[('dorsal-detail',(0,0,1),cv(focus),max(40,dist*4)),('side-detail',(-1,0,0),cv(focus),max(40,dist*4))]
        if v['hull']=='AlienMothership':
            views=[('dorsal-central',(0,0,1),cv((0,88.9,-12)),105),
                   ('ventral-duplicate',(0,0,-1),cv((-196.5,-23.18,96)),100),
                   ('ventral-repair',(0,0,-1),cv((196.5,-23.59,-161.4)),100),
                   ('ventral-mirror-source',(0,0,-1),cv((-196.5,-23.59,-161.4)),100)]
    for view, direction, focus, zoom in views:
        if view=='ventral' and v['hull'] not in ['AlienMothership','AlienDreadnought']:continue
        target=Vector(center) if focus is None else focus
        cam.location=target+Vector(direction)*max(extent*2,500)
        cam.rotation_euler=(target-cam.location).to_track_quat('-Z','Y').to_euler()
        if view.startswith('dorsal'):cam.rotation_euler=(Quaternion((0,0,1),math.pi/2)*cam.rotation_euler.to_quaternion()).to_euler()
        if view.startswith('ventral'):cam.rotation_euler=(Quaternion((0,0,1),-math.pi/2)*cam.rotation_euler.to_quaternion()).to_euler()
        scene.update()
        basis=np.asarray(cam.matrix_world.to_3x3())
        local=(points-center)@basis
        span=local.max(0)-local.min(0)
        camera.ortho_scale=zoom if zoom else max(span[0],span[1]*1600/700)*1.12
        scene.update()
        fname=v['hull'].lower()+'-'+str(v['appearance'])+'-'+view+('-albedo' if textured else '')
        scene.render.filepath=str(OUT/(fname+'-clay.png'))
        bpy.ops.render.render(write_still=True)
        nodes=[]
        for n in v['nodes']:
            if view.startswith(('dorsal','ventral')) and n['side']!=view.split('-')[0]:continue
            if view.startswith('side') and n['side']!='dorsal':continue
            pos=world_to_camera_view(scene,cam,cv(n['position_m']))
            nodes.append(dict(n,pixel=[pos.x*1600,(1-pos.y)*700]))
        if view=='ventral-repair':
            pos=world_to_camera_view(scene,cam,cv((196.5,-23.59000015258789,-161.39999389648438)))
            nodes.append({'side':'ventral','index':18,'ordinary':True,'proposal':True,'pixel':[pos.x*1600,(1-pos.y)*700]})
        render_meta.append({'hull':v['hull'],'appearance':v['appearance'],'view':view,'file':fname+'-clay.png',
                            'nodes':nodes,'ortho_width_m':camera.ortho_scale})
        (TMP/('textured-meta.json' if textured else ('detail-meta.json' if details else 'render-meta.json'))).write_text(json.dumps(render_meta,indent=2))
    print('DONE',v['hull'],v['appearance'],flush=True)

import json,math
from pathlib import Path
import numpy as np
ROOT=Path.cwd(); TMP=ROOT/'.tmp/turret-visual'
scenes=json.loads((TMP/'scene-index.json').read_text())
results=[]
for v in scenes:
 data=np.load(TMP/v['geometry_file']); pieces=[]
 for m in v['meshes']:
  points=data['v'+str(m['key'])][data['t'+str(m['key'])]]
  x=points[:,:,0];z=points[:,:,2]
  denom=(z[:,1]-z[:,2])*(x[:,0]-x[:,2])+(x[:,2]-x[:,1])*(z[:,0]-z[:,2])
  mask=abs(denom)>1e-10
  pieces.append((m['name'],points[mask],denom[mask]))
 def hit(pos,side):
  px,py,pz=pos;hits=[]
  for name,t,d in pieces:
   x=t[:,:,0];z=t[:,:,2]
   a=((z[:,1]-z[:,2])*(px-x[:,2])+(x[:,2]-x[:,1])*(pz-z[:,2]))/d
   b=((z[:,2]-z[:,0])*(px-x[:,2])+(x[:,0]-x[:,2])*(pz-z[:,2]))/d
   ok=(a>=-1e-7)&(b>=-1e-7)&(a+b<=1+1e-7)
   if ok.any():
    y=a[ok]*t[ok,0,1]+b[ok]*t[ok,1,1]+(1-a[ok]-b[ok])*t[ok,2,1]
    hits.append((float(max(y) if side=='dorsal' else min(y)),name))
  if not hits:return None
  yy,name=(max(hits) if side=='dorsal' else min(hits))
  return {'surface_y':yy,'surface_mesh':name,'outward_offset_m':(py-yy)*(1 if side=='dorsal' else -1)}
 for n in v['nodes']:
  if n['side']=='ventral' and v['hull']!='AlienMothership':continue
  results.append({'hull':v['hull'],'appearance':v['appearance'],'side':n['side'],'index':n['index'],
                  'ordinary':n['ordinary'],'position':n['position_m'],'hit':hit(n['position_m'],n['side'])})
 if v['hull']=='AlienMothership':
  candidates=[('mirror_outer_18',(196.5,-23.59000015258789,-161.39999389648438),'ventral'),
              ('central_widen_left',(-9,88.91,3),'dorsal'),('central_widen_right',(9,88.91,3),'dorsal')]
  for xx in (-196.5,196.5):
   for zz in (55,10,-35,-80,-120):
    for side,yy in [('dorsal',25.35),('ventral',-23.59)]:
     candidates.append(('rim_%s_%s_%s'%(xx,zz,side),(xx,yy,zz),side))
  for label,pos,side in candidates:
   samples=[]
   for radius in (0,2,4,6):
    for theta in ([0] if radius==0 else range(0,360,45)):
     p=(pos[0]+radius*math.cos(math.radians(theta)),pos[1],pos[2]+radius*math.sin(math.radians(theta)))
     samples.append({'radius':radius,'angle':theta,'position':p,'hit':hit(p,side)})
   results.append({'hull':v['hull'],'candidate':label,'position':pos,'side':side,'samples':samples})
 print(v['hull'],v['appearance'],flush=True)
(TMP/'surface-probes.json').write_text(json.dumps(results,indent=2),encoding='utf-8')
